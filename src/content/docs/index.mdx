import { Card, CardGrid, LinkCard, Badge } from '@astrojs/starlight/components';
import Architecture from '../../components/diagrams/Architecture.astro';

<CardGrid stagger>
  <Card title="Gateway" icon="seti:json">
    A PHP API for ingesting and reading JSON records, files and high-rate streams. It routes records to typed tables, archives old days to Parquet and serves reads across both tiers. <Badge text="ghcr.io/crowpond/gateway" variant="note" />
  </Card>
  <Card title="Admin console" icon="setting">
    A web console that uses only the gateway's HTTP API: approve schema proposals, browse tables, run the archive, issue API tokens and read the audit trail. <Badge text="ghcr.io/crowpond/admin" variant="note" />
  </Card>
  <Card title="Labelled by design" icon="approve-check">
    Every object carries a STANAG 4774 confidentiality label, bound to the data per STANAG 4778. Archive files get a detached binding with a digest.
  </Card>
  <Card title="Open formats" icon="open-book">
    JSON Schema for payloads, Parquet for the archive, SHA-256 digests, OpenAPI 3.1 for the API, NDJSON logs with W3C trace context.
  </Card>
</CardGrid>
